#### 1. Authentication Service

```typescript
// src/auth/auth.service.ts
import { Injectable } from '@nestjs/common';
import { Clerk } from '@clerk/backend';
import { Cerbos } from '@cerbos/sdk';
import { Redis } from 'ioredis';
import * as jwt from 'jsonwebtoken';

@Injectable()
export class AuthService {
  private readonly clerk: Clerk;
  private readonly cerbos: Cerbos;
  private readonly redis: Redis;

  constructor() {
    this.clerk = new Clerk({ apiKey: process.env.CLERK_API_KEY });
    this.cerbos = new Cerbos({ hostname: process.env.CERBOS_HOST });
    this.redis = new Redis(process.env.REDIS_URL);
  }

  async authenticateUser(token: string) {
    try {
      // Verify Clerk session
      const session = await this.clerk.sessions.verifySession(token);

      // Check if session is valid
      if (!session) {
        throw new UnauthorizedException('Invalid session');
      }

      // Cache user session
      await this.redis.setex(
        `session:${token}`,
        3600, // 1 hour
        JSON.stringify(session)
      );

      return session;
    } catch (error) {
      throw new UnauthorizedException('Authentication failed');
    }
  }

  async authenticateAgent(agentToken: string) {
    try {
      // Verify agent JWT token
      const decoded = jwt.verify(agentToken, process.env.AGENT_JWT_SECRET);

      // Validate agent permissions
      const decision = await this.cerbos.check({
        principal: {
          id: decoded.agentId,
          roles: decoded.roles,
          attr: decoded.attributes,
        },
        resource: {
          kind: 'agent',
          id: decoded.agentId,
        },
        actions: ['access', 'execute'],
      });

      if (!decision.isAllowed('access')) {
        throw new UnauthorizedException('Agent not authorized');
      }

      return decoded;
    } catch (error) {
      throw new UnauthorizedException('Agent authentication failed');
    }
  }

  async authorize(userId: string, resource: string, action: string) {
    const decision = await this.cerbos.check({
      principal: { id: userId },
      resource: { kind: resource, id: '*' },
      actions: [action],
    });

    return decision.isAllowed(action);
  }

  async generateTokens(user: any) {
    const accessToken = jwt.sign(
      {
        sub: user.id,
        email: user.email,
        roles: user.roles,
        type: 'access'
      },
      process.env.JWT_SECRET,
      { expiresIn: '15m' }
    );

    const refreshToken = this.generateSecureToken();

    // Store refresh token
    await this.redis.setex(
      `refresh:${refreshToken}`,
      604800, // 7 days
      user.id
    );

    return { accessToken, refreshToken };
  }

  private generateSecureToken(): string {
    return require('crypto').randomBytes(32).toString('hex');
  }
}
```

#### 2. Authentication Middleware

```typescript
// src/auth/auth.middleware.ts
import { Injectable, NestMiddleware } from '@nestjs/common';
import { Request, Response, NextFunction } from 'express';
import { AuthService } from './auth.service';

@Injectable()
export class AuthMiddleware implements NestMiddleware {
  constructor(private authService: AuthService) {}

  async use(req: Request, res: Response, next: NextFunction) {
    const token = this.extractToken(req);

    if (!token) {
      return res.status(401).json({ message: 'No token provided' });
    }

    try {
      const user = await this.authService.authenticateUser(token);
      req['user'] = user;
      next();
    } catch (error) {
      return res.status(401).json({ message: 'Invalid token' });
    }
  }

  private extractToken(req: Request): string | null {
    const authHeader = req.headers.authorization;
    if (authHeader && authHeader.startsWith('Bearer ')) {
      return authHeader.substring(7);
    }
    return null;
  }
}
```

#### 3. Role-Based Guard

```typescript
// src/guards/roles.guard.ts
import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { ROLES_KEY } from '../decorators/roles.decorator';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const requiredRoles = this.reflector.getAllAndOverride<string[]>(ROLES_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);

    if (!requiredRoles) {
      return true;
    }

    const { user } = context.switchToHttp().getRequest();

    return requiredRoles.some((role) => user.roles?.includes(role));
  }
}
```

```typescript
// src/security/security.service.ts
import { Injectable } from '@nestjs/common';
import * as fs from 'fs';
import * as path from 'path';
import { exec } from 'child_process';
import { promisify } from 'util';
import { Datadog } from 'datadog-api-client';

const execAsync = promisify(exec);

@Injectable()
export class SecurityService {
  private readonly datadog: Datadog;

  constructor() {
    this.datadog = new Datadog({
      apiKey: process.env.DATADOG_API_KEY,
      appKey: process.env.DATADOG_APP_KEY,
    });
  }

  async scanProject(projectPath: string) {
    const results = {
      sast: await this.performSASTScan(projectPath),
      dast: await this.performDASTScan(projectPath),
      dependencies: await this.performDependencyScan(projectPath),
      secrets: await this.performSecretsScan(projectPath),
      infrastructure: await this.performInfrastructureScan(projectPath),
    };

    // Calculate risk score
    const riskScore = this.calculateRiskScore(results);

    // Send to monitoring
    await this.sendSecurityMetrics(results, riskScore);

    return { ...results, riskScore };
  }

  private async performSASTScan(projectPath: string) {
    try {
      // Use Escape.tech for SAST scanning
      const { stdout } = await execAsync(
        `escape scan --type=sast --output=json ${projectPath}`
      );

      const results = JSON.parse(stdout);
      return {
        vulnerabilities: results.vulnerabilities,
        riskLevel: results.riskLevel,
        recommendations: results.recommendations,
      };
    } catch (error) {
      return { error: error.message, vulnerabilities: [] };
    }
  }

  private async performDASTScan(projectPath: string) {
    try {
      // Use Beagle Security for DAST
      const { stdout } = await execAsync(
        `beagle scan --url=http://localhost:3000 --output=json`
      );

      const results = JSON.parse(stdout);
      return {
        vulnerabilities: results.vulnerabilities,
        riskLevel: results.riskLevel,
        endpoints: results.endpoints,
      };
    } catch (error) {
      return { error: error.message, vulnerabilities: [] };
    }
  }

  private async performDependencyScan(projectPath: string) {
    try {
      // Use Trivy for dependency scanning
      const { stdout } = await execAsync(
        `trivy fs --security-checks vuln,config --format json ${projectPath}`
      );

      const results = JSON.parse(stdout);
      return {
        vulnerabilities: results.Results,
        totalVulnerabilities: results.Metadata?.Vulnerabilities?.Count || 0,
        riskLevel: this.calculateDependencyRisk(results),
      };
    } catch (error) {
      return { error: error.message, vulnerabilities: [] };
    }
  }

  private async performSecretsScan(projectPath: string) {
    try {
      // Use GitGuardian for secrets detection
      const { stdout } = await execAsync(
        `gitguardian scan --path=${projectPath} --format=json`
      );

      const results = JSON.parse(stdout);
      return {
        secretsFound: results.secrets.length,
        secrets: results.secrets,
        riskLevel: results.secrets.length > 0 ? 'HIGH' : 'LOW',
      };
    } catch (error) {
      return { error: error.message, secrets: [] };
    }
  }

  private async performInfrastructureScan(projectPath: string) {
    try {
      // Use Checkov for IaC scanning
      const { stdout } = await execAsync(
        `checkov --directory ${projectPath} --framework all --output json`
      );

      const results = JSON.parse(stdout);
      return {
        failedChecks: results.results?.passed_check_count || 0,
        totalChecks: results.summary?.failed + results.summary?.passed || 0,
        riskLevel: this.calculateInfrastructureRisk(results),
      };
    } catch (error) {
      return { error: error.message, checks: [] };
    }
  }

  private calculateRiskScore(results: any): number {
    let score = 0;

    // SAST weight: 30%
    if (results.sast.vulnerabilities.length > 0) {
      score += 30;
    }

    // DAST weight: 25%
    if (results.dast.vulnerabilities.length > 0) {
      score += 25;
    }

    // Dependencies weight: 20%
    if (results.dependencies.totalVulnerabilities > 10) {
      score += 20;
    }

    // Secrets weight: 25%
    if (results.secrets.secretsFound > 0) {
      score += 25;
    }

    return Math.min(score, 100);
  }

  private async sendSecurityMetrics(results: any, riskScore: number) {
    await this.datadog.metric.send('security.scan.risk_score', riskScore);
    await this.datadog.metric.send('security.vulnerabilities.sast', results.sast.vulnerabilities.length);
    await this.datadog.metric.send('security.vulnerabilities.dast', results.dast.vulnerabilities.length);
    await this.datadog.metric.send('security.secrets.found', results.secrets.secretsFound);
    await this.datadog.metric.send('security.vulnerabilities.dependencies', results.dependencies.totalVulnerabilities);
  }
}
```

#### 3. Rate Limiting Middleware

```typescript
// src/middleware/rate.limiter.ts
import { Injectable, NestMiddleware } from '@nestjs/common';
import { Request, Response, NextFunction } from 'express';
import { Redis } from 'ioredis';

@Injectable()
export class RateLimitMiddleware implements NestMiddleware {
  private readonly redis: Redis;
  private readonly windows: Map<string, number> = new Map();

  constructor() {
    this.redis = new Redis(process.env.REDIS_URL);
    this.windows.set('default', 60); // 1 minute
    this.windows.set('login', 15); // 15 minutes
    this.windows.set('api', 100); // 100 requests per minute
  }

  async use(req: Request, res: Response, next: NextFunction) {
    const key = this.getKey(req);
    const windowSize = this.getWindowSize(req.path);

    const current = await this.redis.incr(key);

    if (current === 1) {
      await this.redis.expire(key, windowSize);
    }

    const ttl = await this.redis.ttl(key);

    res.setHeader('X-RateLimit-Limit', this.windows.get('default'));
    res.setHeader('X-RateLimit-Remaining', Math.max(0, this.windows.get('default') - current));
    res.setHeader('X-RateLimit-Reset', Date.now() + ttl * 1000);

    if (current > this.windows.get('default')) {
      return res.status(429).json({
        message: 'Too many requests',
        retryAfter: ttl,
      });
    }

    next();
  }

  private getKey(req: Request): string {
    const ip = req.ip || req.connection.remoteAddress;
    const userAgent = req.headers['user-agent'] || '';
    const hash = require('crypto')
      .createHash('md5')
      .update(`${ip}:${userAgent}`)
      .digest('hex');

    return `rate_limit:${hash}`;
  }

  private getWindowSize(path: string): number {
    if (path.includes('/auth/login')) return this.windows.get('login');
    if (path.startsWith('/api/')) return this.windows.get('api');
    return this.windows.get('default');
  }
}
```

```typescript
class AISecurityManager {
  async validateAgentRequest(request: AgentRequest) {
    // Validate agent identity
    const agentValid = await this.validateAgentIdentity(request.agentId);
    if (!agentValid) {
      throw new UnauthorizedException('Invalid agent identity');
    }

    // Check rate limiting per agent
    const rateLimited = await this.checkAgentRateLimit(request.agentId);
    if (rateLimited) {
      throw new TooManyRequestsException('Agent rate limit exceeded');
    }

    // Validate request payload
    const payloadValid = await this.validatePayload(request.payload);
    if (!payloadValid) {
      throw new BadRequestException('Invalid request payload');
    }

    // Check for prompt injection
    const injectionDetected = await this.detectPromptInjection(request.payload);
    if (injectionDetected) {
      throw new BadRequestException('Potential injection detected');
    }

    return true;
  }

  private async detectPromptInjection(payload: string): Promise<boolean> {
    const injectionPatterns = [
      /\b(ignore|override|bypass|sudo|admin|root)\b/i,
      /\b(exec|eval|system|shell)\b/i,
      /\b(script|javascript|vbscript)\b/i,
      /\[.*\]\(.*\)/, // Markdown injection attempt
    ];

    return injectionPatterns.some(pattern => pattern.test(payload));
  }
}
```

```typescript
    return this.pqc.sign(data, privateKey);
  }

  async verify(data: Buffer, signature: string, publicKey: string): Promise<boolean> {
    return this.pqc.verify(data, signature, publicKey);
  }
}
```
